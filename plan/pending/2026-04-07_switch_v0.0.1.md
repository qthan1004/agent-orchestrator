# Implementation Plan: `@thanh-libs/switch`

## Tổng quan

Toggle on/off switch — Tier 2 Foundation, chỉ phụ thuộc `theme` + `utils`.
Dùng `<button role="switch">` theo WAI-ARIA Switch Pattern.

---

## API Design

> Tham khảo MUI Switch, Ant Design Switch, shadcn/ui Switch.

### Props

```typescript
export type SwitchSize = 'sm' | 'md' | 'lg';
export type SwitchColor = 'primary' | 'secondary' | 'error' | 'warning' | 'info' | 'success';

export interface SwitchProps extends Omit<HTMLAttributes<HTMLButtonElement>, 'onChange'> {
  /** Controlled checked state */
  checked?: boolean;

  /** Default checked for uncontrolled mode */
  defaultChecked?: boolean;

  /** Change handler — receives new checked value */
  onChange?: (checked: boolean) => void;

  /** Disabled state */
  disabled?: boolean;

  /** Switch size (default: 'md') */
  size?: SwitchSize;

  /** Theme color when checked (default: 'primary') */
  color?: SwitchColor;

  /** Accessible label (required when no external label) */
  label?: string;

  /** Place label before or after (default: 'end') */
  labelPlacement?: 'start' | 'end';

  /** Additional CSS class */
  className?: string;
}
```

### Controlled vs Uncontrolled

```tsx
// Controlled
const [on, setOn] = useState(false);
<Switch checked={on} onChange={setOn} />

// Uncontrolled
<Switch defaultChecked />
```

**Styling details:**
- **Track off:** `palette.action.disabled` background, `border-radius: height/2`
- **Track on:** `palette[color].main` background
- **Thumb:** white circle, `box-shadow: shadows[1]`, absolutely positioned
- **Thumb on:** `transform: translateX(travel)` with `transition: 0.2s ease`
- **Disabled:** `opacity: 0.5`, `cursor: not-allowed`
- **Focus-visible:** `outline: 2px solid palette[color].main` on track, `outline-offset: 2px`

### Component

#### [NEW] `src/lib/Switch.tsx`

```tsx
export const Switch = forwardRef<HTMLButtonElement, SwitchProps>(
  ({
    checked: controlledChecked,
    defaultChecked = false,
    onChange,
    disabled,
    size = SWITCH_DEFAULTS.size,
    color = SWITCH_DEFAULTS.color,
    label,
    labelPlacement = SWITCH_DEFAULTS.labelPlacement,
    className,
    ...rest
  }, ref) => {
    // Controlled vs Uncontrolled
    const [internalChecked, setInternalChecked] = useState(defaultChecked);
    const isControlled = controlledChecked !== undefined;
    const isChecked = isControlled ? controlledChecked : internalChecked;

    const handleClick = () => {
      if (disabled) return;
      const next = !isChecked;
      if (!isControlled) setInternalChecked(next);
      onChange?.(next);
    };

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        handleClick();
      }
    };

    return (
      <SwitchRootStyled
        ownerLabelPlacement={labelPlacement}
        ownerDisabled={disabled}
        className={className}
      >
        <SwitchTrackStyled
          ref={ref}
          role="switch"
          type="button"
          aria-checked={isChecked}
          aria-disabled={disabled || undefined}
          aria-label={!label ? rest['aria-label'] : undefined}
          ownerChecked={isChecked}
          ownerSize={size}
          ownerColor={color}
          ownerDisabled={disabled}
          onClick={handleClick}
          onKeyDown={handleKeyDown}
          tabIndex={disabled ? -1 : 0}
          {...rest}
        >
          <SwitchThumbStyled ownerChecked={isChecked} ownerSize={size} />
        </SwitchTrackStyled>
        {label && (
          <SwitchLabelStyled ownerSize={size} ownerDisabled={disabled}>
            {label}
          </SwitchLabelStyled>
        )}
      </SwitchRootStyled>
    );
  }
);

Switch.displayName = 'Switch';
```
